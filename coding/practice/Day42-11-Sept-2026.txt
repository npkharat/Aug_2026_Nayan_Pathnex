## Ansible — Create Docker Containers for Multiple Applications

- name: Setup docker containers for multiple application 
  hosts: all
  become: yes
  tasks: 
    - name: Install dependencies
      yum:
        name: "python3-pip"
        state: present
    - name: Install docker
      pip:
        name: docker
        state: present
    - name: Pull web-app image
      docker_image:
        name: my-web-app
        source: pull
    - name: pull DB app image
      docker_image:
        name: my-db-app
        source: pull
    - name: run web app container
      docker_container:
        name: web-app
        image: my-web-app
        state: started
        published_ports:
            - "8080:80"
    - name: run db container
      docker_container:
        name: db-server
        image: my-db-app
        state: started
        published_ports:
            - "3306:3306"

##Kubernetes — Deploying StatefulSet

apiVersion: v1
kind: StatefullSet
metadata:
    name: pathnex-db
spec:
    replicas: 3
    serviceName: "pathnex-db"
    selector:
        matchLabels:
            app: pathnex-db
    template:
        metadata:
            labels:
                app: pathnex-db
        spec:
            containers:
                name: pathnex-db
                image: mysql:5.7
                volumeMounts:
                    - mountPath: /var/lib/mysql
                      name: my-sql-storage
    volumeClaimTemplates:
        - metadata:
            name: mysql-storage
          spec:
            accessModes: ["ReadWriteOnce"]
            resources:
                requests:
                    storage: 1Gi

##Jenkinsfile — Build and Deploy to Kubernetes
pipeline{
    agent any
    environment {
        IMAGE_NAME = 'pathnex-web-app'
    }

    stages{
        stage('Build image'){
            steps{
                script{
                    docker.build($"env.IMAGE_NAME")
                }
            }
        }
        stage('Push image'){
            steps{
                script{
                    docker.withRegistry('https://docker.io', 'docker-credentials'){
                        docker.image("${env.IMAGE_NAME}").push('latest')
                    }
                }
            }
        }
        stage('Deploy to k8s'){
            steps{
                script{
                    sh 'kubectl apply -f kubernetes/deployment.yaml'
                }
            }
        }
    }
