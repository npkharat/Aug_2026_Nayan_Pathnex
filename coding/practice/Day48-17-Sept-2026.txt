Ansible — Create and Start EC2 Instances

- name: Create and start ec2
  hosts: localhost
  become: true
  tasks:
    - name: create ec2 in us-east-1
      ec2_instance:
        name: pathnex-ec2-us-east-1
        key_name: pathnex-key
        region: us-east-1
        instance_type: t2.micro
        image: ami-xxxxxxxxxxxxxxx
        wait: yes
    - name: Create instance in us-west-1
      ec2_instance:
        name: instance-us-west
        key_name: pathnex-key
        region: us-west-1
        instance_type: t2.micro
        image: ami-xxxxxxxxxxxxxxx
        wait: yes

##Kubernetes — Multi-Cluster Deployment with Federation
kind: Deployment
apiVersion: apps/v1
metadata:
    name: pathnex-deployment
spec:
    replicas: 2
    selector:
        matchLabels:
            app: pathnex-app
    template:
        metadata:
            labels:
                app: pathnex-app
        spec:
            containers:
                - name: app
                  image: nginx
                  ports:
                    - containerPort: 80
---
kind: Service
apiVersion: v1
metadata:
    name: pathnex-service
spec:
    selector:
        app: pathnex-app
    ports:
        - protocol: TCP
          port: 80

##Jenkinsfile — Deploy to Multi-Cluster Environments
pipeline{
    agent any
    environment{
        CLUSTER = "us-east-1"
    }
    stages{
        stage('Build'){
            steps{
                script{
                    docker.build('pathnex-web-app')
                }
            }
        }
        stage('test'){
            steps{
                sh 'echo testing'
            }
        }
        stage('Deploy'){
            steps{
                script{
                    if (env.CLUSTER == 'us-east-1'){
                        sh'''
                        kubectl --context=us-east-1 apply -f kubernetes/deployment.yaml
                    }
                    else{
                        sh'''
                        kubectl --context=us-west-1 apply -f kubernetes/deployment.yaml
                    }
                }
            }
        }
    }
}

#GitLab CI/CD — Multi-Region Deployment with Kubernetes
stages:
    - build
    - push
    - deploy
build:
    stage: build
    script:
     - docker build -t pathnex-web-app .
push:
    stage: push
    script:
        - docker login -u "$USERNAME" -p "$PASSWORD"
        - docker push pathnex-web-app
deploy:
    stage: deploy
    script:
        - |
          if [ $COMMMIT_REF == "main" ];
          then
            kubectl --context=us-east-1 apply -f kubernetes/prod-deployment.yaml
          else
            kubectl --context=us-west-1 apply -f kubernetes/dev-devlopment.yaml
        fi
