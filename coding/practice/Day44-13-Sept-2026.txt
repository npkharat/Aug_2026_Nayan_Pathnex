Day 12 — Scaling Applications and Automation

#Ansible — Setup Auto-Scaling Group (ASG)
- name: Setup auto scaling Group
  hosts: all
  become: yes
  tasks:
    - name: create launch configuration
      ec2_asg:
        name: pathnex-asg
        region: us-east-1
        launch_config_name: pathnex-launch-config
        min_size: 1
        max_size: 3
        desired_capacity: 2
        vpc_zone_identifier: subnet-xxxxxxxxxxx

#Terraform — Create a Load Balancer with EC2 Instances

resource "aws_lb" "pathnex_lb"{
    name = "pathnex-lb"
    internal = false
    load_balancer_type = "application"
    security_group = [aws_security_group.allow_ssh_http.id]
    subnet = [aws_subnet.public.id]
}

resource "aws_instance" "pathnex_ec2"{
    ami = "ami-xxxxxxxxxxx"
    instance_type = "t2.micro"
    security_groups = [aws_security_group.allow_ssh_http.name]
    tags = {
        Name = "pathnex-ec2"
    }
}


#Kubernetes — Horizontal Pod Autoscaler (HPA) with Load Balancer
kind: Deployment
apiVersion: apps/v1
metadata:
    name: pathnex-web-app-deployment
spec:
    repicas: 2
    selector:
        matchLabels:
            app: pathnex-web-app
    template:
        metadata:
            labels:
                app: pathnex-web-app
        spec:
            containers:
                - name: app
                  image: nginx
                  ports:
                    - containerPort: 80

---
kind: HorizontalPodAutoscaler
apiVersion: autoscaling/v2
metadata:
    name: pathnex-web-app-hpa
spec:
    scaleTargetRef:
        apiVersion: apps/v1
        kind: Deployment
        name: pathnex-web-app-deployment
    minReplicas: 2
    maxReplicas: 10
    targetCPUUtilizationPercentage: 70

#Jenkinsfile — Implement Parallel Stages
pipeline {
    agent any
    stages{
        stage('Build & test'){
            parallel{
                stage('Checkout & Build'){
                    steps{
                        git branch: 'main' , url: 'https://www.github/sample_node.git_url'

                        sh'''
                        cd sample_node
                        npm install
                        npm run build
                        '''
                    }
                }
                stage('test'){
                    steps{
                        sh'''
                        npm test || {echo "test failed"; exit 1;}
                        '''
                    }
                }
            }
            stage('Deploy'){
                steps{
                    sh 'echo "Deploying application"'
                }
            }
        }
    }
}
