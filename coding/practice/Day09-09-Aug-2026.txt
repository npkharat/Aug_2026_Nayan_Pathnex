#Jenkinsfile — Basic CI Pipeline
pipeline{
    agent any
    stages{
        stage('checkout'){
            steps{
                git branch: 'main', utl: 'https_xyzExample.git'
            }
        }
        stage('build'){
            steps{
                sh'''
                echo "App building"
                '''
            }
        }
        stage('test'){
            steps{
                sh'''
                echo "App testing"
                '''
            }
        }
        stage('package'){
            steps{
                sh'''
                echo "app package"
                '''
            }
        }
    }
}

#GitLab CI/CD — Basic Pipeline
stages:
    - build
    - test
    - deploy
build:
    stage: build
    script:
        - echo "Building app"

test:
    stage: test
    script:
        - echo "app testing"

deploy:
    stage: deploy
    script:
        - echo "deploying app"

#Ansible — Install and Configure Nginx
- name: Install and configure nginx
  hosts: all
  become: yes
  tasks: 
    - name: install nginx
      yum:
        name: nginx
        state: present
    - name: start nginx
      service:
        state: started
        enabled: yes
    - name: configure nginx.conf
      template:
        src: nginx.conf.j2
        dest: /etc/nginx/nginx.config
    - name: restart nginx
      service:
        state: restarted
    
#Terraform — EC2 with Security Group
provider "aws" {
    region = "ap-south-1"
}

resource "aws_security_group" "pathnex_sg"{
    name = "pathnex-sg"
    description = "sg to enable ssh and http"

    ingress {
        from_port = 22
        to_port = 22
        protocol = 'tcp'
        cidr = ["0.0.0.0/0"]
    }

    ingress{
        from_port = 80
        to_port = 80
        protocol = 'tcp'
        cidr = ["0.0.0.0/0"]
    }

    egress {
        from_port = 0
        to_port 0
        protocol = '-1'
        cidr = ["0.0.0.0/0"]
    }
}

resource "aws_instance" "pathnex_ec2"{
    ami = "ami-0529"
    instance_type = "t3.medium"
    vpc_security_group_ids = [aws_security_group.pathnex.sg.id]
}
