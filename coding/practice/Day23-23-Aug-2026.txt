    }

                docker.build('pathnex-web-app')
                }
            }
        }
        stage('Test'){
            steps{
                sh'''
                echo "testing...."
                '''
            }
        }

        stage('Deploy to Kubernetes'){
            steps{
                script{
                    if (env.CLUSTER == 'us-east-1'){
                        sh'''
                        kubectl --context=us-east-1 apply -f kubernetes/deployment.yaml
                        '''
                    } 
                    else{
                        sh'''
                        kubectl --context=us-west-1 apply -f kubernetes/deployment.yaml
                    }
                }
            }
        }
    }
}

# Day 17 — Multi-Cloud Deployments, Auto-healing, and Advanced CI/CD

## Ansible — Auto-healing Web Servers with Systemd
- name: configure Auto-healing for webservers
  hosts: all
  become: yes
  tasks:
    - name: ensure ngnix is running
      service:
        name: ngnix
        state: started
        enabled: yes
    - name: auto start for ngnix service
      lineinfile:
        path: /etc/systemd/system/ngnix.service.d/override.conf 
        line: "Restart=always"
        create: yes
    - name: Reload ngnix
      command: systemctl demon-reload

##Kubernetes — Self-Healing with Liveness and Readiness Probes
kind: Deployment
apiVersion: apps/v1
metadata:
    name: pathnex-web-app
spec:
    replicas: 2
    selector:
        matchLabels:
            app: pathnex-web-app
    template:
        metadata:
            labels:
                app: pathnex-web-app
        spec:
            containers:
                - name: app
                  image: ngnix
                  ports:
                    containerPort: 80
                  livenessProbe:
                    httpGet:
                        path: /healthz
                        port: 80
                    initialDelaySeconds: 5
                    periodSeconds: 10
                  readinessProbe:
                    htteGet:
                        path: /healthz
                        port: 80
                    initialDelaySeconds: 5
                    periodSeconds: 10

## Ansible — Configure HAProxy for Load Balancing
- name: setup HAproxy load balancer
  hosts: all
  become: yes
  tasks:
    - name: Install HAproxy
      yum: 
        name: haproxy 
        state: present
    - name: Configure HAproxy
      template:
        src: haproxy.cfg.j2
        dest: /etc/haproxy/haproxy/cfg
    - name: start HAproxy service
      service: 
        name :haproxy
        state: started
        enabled: yes
