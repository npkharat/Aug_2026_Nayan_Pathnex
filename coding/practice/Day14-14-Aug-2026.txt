#Ansible — Multi-Node Nginx Setup
- name: setup nginx on multiple nodes
  hosts: all
  become: yes
  tasks:
    - name: Install nginx
      yum:
        name: nginx
        state: present
    - name: Start nginx
      service:
        state: started
        enabled: yes

#Terraform — Create IAM Users and Policies

}

resource "aws_iam_policy" "example_policy"{
    name = "pathnex-policy"
    description = "policy for pathnex application"
    policy = jsonencode({
        Version = "2012-10-17"
        Statement = [{
            Action = "s3:ListBucket"
            Effect = "Allow"
            Resource = "arn:aws:s3:::pathnex-bucket"
        }]
    })
}

#Kubernetes — Deploy Nginx with ConfigMap and Secrets
kind: ConfigMap
apiVersion: v1
metadata:
    name: nginx-config
data:
    nginx.conf: |
        server{
            listen 80;
            server_name localhost;
            location / {
                root /usr/share/nginx/html;
                index index.html index.html;
            }
        }

---

apiVersion: v1
kind: Secret
metadata:
    name: nginx-secret
type: Opaque
data:
    password: abcdefghi

---

kind: Deployment
apiVersion: apps/v1
metadata:
    name: pathnex-nginx
spec:
    replicas: 2
    selector:
        matchLabels:
            app: pathnex-nginx
    templete:
        metadata:
            labels:
                app: pathnex-nginx
        spec:
            containers:
                - name: nginx
                  image: nginx
                  volumeMounts:
                    - mountPath: /etc/nginx/ngnix.conf
                      name: nginx-config
                      subPath: nginx.conf
            volumes:
                - name: nginx-config
                  configMap: 
                    name: nginx-config
                - name: nginx-secret
                  secret:
                    secretName: nginx-secret

#Jenkinsfile — Multi-Branch Pipeline with Docker Build

pipeline{
    agent any
    stages{
        stage('Build'){
            steps{
                script{
                    docker.build('pathnex-app')
                }
            }
        }
        stage('Deploy'){
            steps{
                script{
                    kubectl apply -f kubernetes/deployment.yaml
                }
            }
        }
    }
}

#GitLab CI/CD — Advanced Deployment with Helm
stages:
    - build
    - push
    - deploy

build:
    stage: build
    script:
        - docker build -t pathnex-web-app .

push:
    stage: push
    script:
        - docker login -u "$USERNAME" -p "$PASSWORD"
        - docker push pathnex-web-app

deploy:
    stage: deploy
    script:
        - helm repo add pathnex https://charts.pathnex.com
        - helm install pathnex-nginx pathnex/nginx-ingress
