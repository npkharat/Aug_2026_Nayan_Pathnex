#Ansible — Setup Docker Container for Nginx
- name: Setup Docker Container for Nginx
  hosts: all
  become: yes
  tasks: 
    - name: install docker
      yum:
        name: docker
        state: present
    - name: Pull nginx image
      docker_image:
        name: nginx
        source: pull
    - name: Run nginx Container
      docker_container:
        name: nginx-container
        image: nginx
        state: started
        published_ports:
            - "8080:80"

##Ansible — Install Helm on Server
- name: Install Helm chart
  hosts: all
  become: yes
  tasks: 
    - name: download Helm
      get_url:
        url: "https://get.helm.sh/helm-v3.7.0-linux-amd64.tar.gz" 
        dest: /tmp/helm.tar.gz
    - name: extract helm package
      unarchive:
        src: /tmp/helm.tar.gz
        dest: /usr/local/bin/
        remote_src: yes
    - name: Set helm permission binaries
      file:
        path: /usr/local/bin/helm
        mode: '0755'

##Ansible — Setup Nginx with Reverse Proxy

  hosts: all
  become: yes
  tasks:
    - name: Install nginx
      yum:
        name: nginx
        state: present
    - name: copy ngnix.config file

        src: reverse-proxy.conf.j2

    - name: Restart ngnix
      service:
        name: nginx
        state: restarted
        enabled: yes

#Terraform — EC2 with Security Group and EBS Volume
provider "aws"{
    region = "ap-south-1"
}
resource "aws_security_group" "pathnex_sg_allow_ssh_http"{
    name = "allow_ssh_http"
    description = "Allow SSH and HTTP ports"

    ingress{
        description = "ssh"
        from_port = 22
        to_port = 22
        protocol = "tcp"
        cidr_block = "[OUR_PUBLIC_ID/32]"
    }

    ingress{
        description = "http"
        from_port = 80
        to_port = 80
        protocol = "tcp"
        cidr_block = ["OUR_PUBLIC_ID/32"]
    }

    egress{
        from_port = 0
        to_port = 0
        protocol = "-1"
        cidr_block = ["0.0.0.0/0"]
    }
}
