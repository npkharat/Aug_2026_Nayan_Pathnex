# Parallel Builds
## 🔹 Jenkins Pipeline — Parallel Build & Test

pipeline{
    agent any
    stages{
        stage('Parallel task'){
            parallel{
                stage('Build'){
                    steps{
                        sh 'mvn clean compile'
                    }
                }
                stage('Test){
                    steps{
                        sh 'mvn test'
                    }
                }
            }
        }
    }
}

## 🔹 GitLab CI — Parallel Jobs
stages:
    - build-test

maven-build:
    stage: build-test
    image: maven:3.8.1-jdk-17
    script:
        - git clone https://github.com/Pathnex/sample-java-app.git
        - cd smaple-java-app
        - mvn clean compile
    parallel: 1

maven-test:
    stage: build-test
    image: maven:3.8.1-jdk-17
    scrips:
        - cd smaple-java-app
        - mvn test
    parallel: 1

🔹 Ansible — Install and Configure Nginx

---
  - name: Install ansible & configure nginx
    hosts: all
    become: yes
    tasks:
        - name: install nginx
          yum: 
            name: nginx
            state: present
        - name: Configure nginx
        template:
            src: nginx.conf.j2
            dest: /etc/nginx/nginx.conf
        
        - name: start nginx
          service:
            name: nginx
            state: started
            enabled: yes
