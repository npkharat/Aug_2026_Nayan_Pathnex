## Ansible — Auto-healing Web Servers with Systemd

- name: cnfigure auto-scaling for webservers
  hosts: all
  become: yes
  tasks:
    - name: Ensure nginx is running
      service:
        name: nginx
        state: started
        enabled: yes
    - name: auto start for nginx service
      lineinfile:
        path: /etc/systemd/system/nginx.service.d/override.config 
        line: "Restart=always"
        create: yes
    - name: reload nginx
      command: systemctl demon-reload

##Kubernetes — Self-Healing with Liveness and Readiness Probes

kind: Deployment
apiVersion: apps/v1
metadata:
    name: pathnex-web-app
spec:
    replicas: 2
    selector:
        matchLabels:
            app: pathnex-web-app
    template:
        metadata:
            labels:
                app: pathnex-web-app
        spec:
            containers:
                - name: app
                  image: nginx
                  ports:
                    - containerPort: 80
                  livenessProbe:
                    httpGet:
                        path: /health
                        port: 80
                    initialDelaySeconds: 5
                    periodSeconds: 10
                  readinessProbe:
                    htteGet:
                        path: /health
                        port: 80
                    initialDelaySeconds: 5
                    periodSeconds: 10

##Jenkinsfile — Rolling Deployments with Kubernetes
pipeline{
    agent any
    environment{
        IMAGE_NAME = 'pathnex-web-app'
    }
    stages{
        stage('Build'){
            steps{
                sh'''
                echo "Building docker image"
                docker build -t "${IMAGE_NAME}"
            }
        }
        stage('Test'){
            steps{
                script{
                    sh'''
                    echo "Running tests"
                    '''
                }
            }
        }
        stage('Deply'){
            steps{
                scripts{
                    sh'''
                    kubectl apply -f kubernetes/deployment.yaml
                    kubectl rollout status deployment/pathnex-web-app
                    '''
                }
            }
        }
    }
}
