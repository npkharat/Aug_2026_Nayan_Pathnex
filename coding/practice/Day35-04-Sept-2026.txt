##Ansible — Install & Enable Nginx

- name: Install and enable ngnix
  hosts: all
  become: yes
  tasks:
    - name: install ngnix
      yum:
        name: ngnix
        state: present
    - name: enable ngnix
      service:
        name: ngnix
        state: started
        enabled: yes

##Kubernetes — Deployment with 2 Replicas

    name: pathnex-deployment

    replicas: 2

    template:

## Jenkins Pipeline — Maven Build
pipeline {
    agent any
    tools {
        maven 'Maven-3.8.1'
        jdk 'JDK-17'
    }

    stages{
        stage('checkout'){
            steps{
                git branch: 'main', url: 'https://github.com/Pathnex/sample-java-app.git'
            }
        }
        stage('Compile'){
            steps{
                sh'''
                cd sample-java-app
                mvn clean compile
                '''
            }
        }
        stage('test'){
            steps{
                sh'''
                cd sample-java-app
                mvn test
                '''
            }
        }
        stage('package'){
            steps{
                sh'''
                cd sample-java-app
                mvn run build
                '''
            }
        }
    }
}

##GitLab CI — Maven Build
stages:
    - build
    - test
    - package
build:
    stage: build
    image: maven:3.8.1-jdk-17
    script:
        - git clone https://github.com/Pathnex/sample-java-app.git
        - cd sample-java-app
        - mvn clean compile

test:
    stage: test
    image: maven:3.8.1-jdk-17
    script:
        - cd sample-java-app
        - mvn test

package:
    stage: package
    image: maven:3.8.1-jdk-17
    script:
        - cd sample-java-app
