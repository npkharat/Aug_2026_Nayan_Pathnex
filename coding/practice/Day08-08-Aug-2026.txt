##Ansible — Handler to Restart Nginx
- name: manage nginx
  hosts: all
  become: yes
  tasks:
    - name: udpate ngnix file
      copy:
        src: index.html
        dest: /usr/share/nginx/html/index.html
      notify: Restart Nginx

  handlers:
    - name: Restart Nginx
      service:
        name: nginx
        state: restarted
