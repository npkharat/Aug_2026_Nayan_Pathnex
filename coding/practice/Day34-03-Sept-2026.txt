##Ansible Task — Install Nginx on Pathnex Server

- name: Install nginx and pathnex server
  hosts: all
  become: yes
  tasks:
    - name: install nginx
      yum:
        name: nginx
        state: present

##Kubernetes Task — Create Nginx Pod

apiVersion: v1
metadata:
    name: pathnex-pod
spec:
    containers:
        - name: pathnex-nginx
          image: ngnix:latest
          ports:

##Jenkins Pipeline — Checkout Git
pipeline{
    agent any
    stages{
        stage('Checkout'){
            steps{
                git branch: 'main' url: 'https://github.com/Pathnex/sample-java-app.git'
            }
        }
        stage('List files'){
            steps{
                sh'''
                cd sample-java-app
                ls -la
                '''
            }
        }
    }
}

##GitLab CI — Checkout Git
stages:
    - checkout
git-checkout:
    stage: checkout
    script:
        - git clone https://github.com/Pathnex/sample-java-app.git
        - cd sample-java-app
        - ls -la
