Ansible - 
- name: Setup Ngnix with custom config
  hosts: all
  become: yes
  tasks:
    - name: Install ngnix
      yum:
        name: ngnix
        state: present
    - name: copy customer ngnix.config
      copy:
        src: ngnix.config
        dest: /etc/ngnix/ngnix/config
    - name: start ngnix service
      service:
        name: ngnix
        state: started
        enabled: yes
