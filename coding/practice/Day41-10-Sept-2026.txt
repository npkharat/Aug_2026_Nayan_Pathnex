##Terraform — EC2 with Security Group and EBS Volume

provider "aws" {
    region = "ap-south-1"
}

resource "aws_security_group" "pathnex_sg" {
    name = "allow_ssh_http"
    description = "allow ssh & http"

    ingress{
        from_port = 22
        to_port = 22
        protocol = "tcp"
        cidr_block = ["0.0.0.0/0"]
    }

    ingress {
        from_port = 80
        to_port = 80
        protocol = "tcp"
        codr_block = ["0.0.0.0/0"]
    }
}

resource "aws_instance" "pathnex_ec2" {
    ami = "ami-xxxxxxxxxxxxxxxxxxx"
    instance_type = "t2.micro"
    aws_security_group_ides = [aws_security_group.pathnex_ec2.id]
    ebs_block_device = {
        device_name = "/dev/sdh"
        volume_size = 30
    }
}

##Kubernetes — Service with NodePort
kind: Service
apiVersion: v1
metadata:
    name: pathnex-service
spec:
    type: NodePort
    selector:
        app: pathnex-app
    ports:
        - protocol: TCP
          port: 80
          nodePort: 30007

##Jenkinsfile — Docker Build and Push
pipeline{
    agent any
    stages{
        stage('Build'){
            steps{
                sh'''
                docker build -t pathnex-app .
                '''
            }
        }
        stage('Push image to docker registry){
            steps{
                script{
                    docker.withRegistry('https://docker.io', 'docker-credentials'){
                        docker.image('pathnex-app').push('latest')
                    }
                }
            }
        }
    }
}

##GitLab CI/CD — Build & Push Docker Image
stages:
    - build
    - push
image-build:
    stage: build
    script:
        - docker build -t pathnex-app .
image-push:
    stage: push
    script:
        - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
        - docker push pathnex-app
