## Terraform — Security Group (r6i.4xlarge EC2)
provider "aws" {
    region = "ap-south-1"
}
resource "aws_security_group" "pathnex_sg_http_ssh"{
    name = "pathnex-sg-allow-ssh-http"
    ingress{
        from_port = 22
        to_port = 22
        protocol = "tcp"
        cidr_block = ["0.0.0.0/0"]
    }

    ingress{
        from_port = 80
        to_port = 80
        protocol = "tcp"
        cidr_block = ["0.0.0.0/0"]
    }

    egress{
        from_port = 0
        to_port = 0
        protocol = "-1"
        cidr_block = ["0.0.0.0/0"]
    }
}

resource "aws_instance" "pathnex_ec2"{
    ami = "ami-xxxxxxxxxxxxxxxx"
    instance_type = "r6i.4xlarge"
    vpc_security_group_ids = [aws_security_group.pathnex_sg_http_ssh.id]
    tags = {
        Name = "pathnex-ec2"
        Environment = "Pathnex-dev"
        Owner = "Pathnex-nayan"
    }
}

## Jenkins Pipeline — NodeJS Build

pipeline{
    agent any
    tools {
        nodejs "NodeJS-16"
    }
    stages{
        stage('Checkout'){
            steps{
                git url: 'https://github.com/Pathnex/sample-node-app.git'
            }
        }
        stage('Install Dependancies'){
            steps{
                sh'''
                cd sample-node-app
                npm install
                '''
            }
        }
        stage('Run test'){
            steps{
                sh'''
                npm test
                '''
            }
        }
        stage('Build'){
            steps{
                sh'''
                npm run build
                '''
            }
        }
    }
}
