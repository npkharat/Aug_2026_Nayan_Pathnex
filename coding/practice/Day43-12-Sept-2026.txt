#Ansible — Install Helm on Server

- name: Install Helm
  hosts: all
  become: yes
  tasks:
    - name: dpwnload helm
      get_url:
        url: https://get.helm.sh/helm-v3.7.0-linux-amd64.tar.gz
      dest: /tmp/helm.tar.gz
    
    - name: unzip helm
      unarchive:
        src: /tmp/helm.tar.gz
        dest: /usr/local/bin/
        remote_src: yes
    - name: set helm permissions
      file:
        path: /usr/local/bin/helm
        mode: '0755'

#Jenkinsfile — Multi-Stage Pipeline
pipeline{
    agent any
    environment {
        IMAGE_NAME = "pathnex-web-app"
    }

    stages{
        stage('Build image'){
            steps{
                script{
                    docker.build("${env.IMAGE_NAME}")
                }
            }
        }
        stage('push image'){
            steps{
                script{
                    docker.withRegistry('https://docker.io', 'docker-credintials'){
                        docker.image("${env.IMAGE_NAME}").push('latest')
                    }
                }
            }
        }

        stage('Deploy to k8s'){
            steps{
                script{
                    sh 'kubectl apply -f kubernetes/deployment.yaml'
                }
            }
        }
    }
}

#GitLab CI/CD — Build, Push, and Deploy with Helm
stages:
    - build
    - push
    - deploy
build:
    stage: build
    script:
        - git clone https://github.com/pathnex/sample-repo.git
        - cd sample-repo
        - docker build -t pathnex-web-app .
push:
    stage: push
    script:
        - docker login -u "${CI_USER}" -p "${CI_REGISTRY_PASSWORD}"
        - docker push pathnex-web-app

deploy:
    stage: deploy
    script:
        - kubectl apply -f pathnex-ngnix/ngnix-ingress.yaml
